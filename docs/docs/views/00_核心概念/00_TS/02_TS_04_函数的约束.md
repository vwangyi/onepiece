## 函数重载

```ts
function func(a: number, b: number): number;
function func(a: string, b: string): string;
function func(a: string | number, b: number | string): number | string {
    if(typeof a === 'string' && typeof b === 'string') {
        return a + b;
    }
    if(typeof a === 'number' && typeof b === 'number') {
        return a * b;
    }
    throw new Error('a和b必须是一样的类型')
}

func(1,2)
func('a','b')

```

## 可选参数和默认参数
- 可选参数必须在参数列表的末尾
```ts
// 可选参数 就是类型加了 | undefined
function func(a: number, b?: number, c: number = 0) {
    if (b) {
        return a + b
    }
    else {
        return a;
    }
}
```
